📊 เชื่อมต่อ Google Sheets & โค้ด Apps Script (doPost / doGet)

รูปแบบการส่งข้อมูล:
💻 ๒. โค้ด Google Apps Script (รองรับทั้ง doPost(e) และ doGet(e) ในตัวเดียว) 
/**
 * โค้ด Google Apps Script สำหรับระบบลอยกระทงออนไลน์ ๒๕๖๙ (เก็บคำอธิษฐานพนักงาน)
 * รองรับทั้ง doPost(e) และ doGet(e) เพื่อรับข้อมูลจากหน้าเว็บแล้วบันทึกลงแถวใหม่ของ Google Sheets
 */
const SHEET_NAME = 'คำอธิษฐานพนักงาน';

// 1. ฟังก์ชันรับข้อมูลแบบ POST จากหน้าเว็บ (doPost)
function doPost(e) {
  return handleRequest(e, 'POST');
}

// 2. ฟังก์ชันรับข้อมูลและดึงข้อมูลแบบ GET จากหน้าเว็บ (doGet)
function doGet(e) {
  // ถ้าส่งพารามิเตอร์ action=insert หรือมีข้อมูล wish ส่งมาทาง URL ให้บันทึกลงแถวใหม่ทันที
  if (e && e.parameter && (e.parameter.action === 'insert' || e.parameter.wish)) {
    return handleRequest(e, 'GET');
  }
  // ถ้าเป็นการเรียกดูข้อมูล (action=read) ให้ดึงรายการคำอธิษฐานล่าสุดกลับไปแสดงบนหน้าเว็บ
  return getRecentWishes();
}

function handleRequest(e, methodUsed) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000); // ป้องกันข้อมูลชนกันกรณีพนักงานกดลอยกระทงพร้อมกันหลายคน

  try {
    const sheet = getOrCreateSheet();
    let data = {};

    // อ่านข้อมูลจาก JSON Body (กรณีส่งแบบ fetch POST)
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (err) {
        data = e.parameter || {};
      }
    } else if (e && e.parameter) {
      // อ่านข้อมูลจาก URL Query / Form Parameter (กรณีส่งแบบ doGet หรือ Form)
      data = e.parameter;
    }

    const timestamp = Utilities.formatDate(new Date(), 'Asia/Bangkok', 'dd/MM/yyyy HH:mm:ss');
    const name = data.name || 'พนักงานผู้มีจิตศรัทธา';
    const department = data.department || 'ไม่ระบุแผนก';
    const krathongName = data.krathongName || data.krathongType || 'กระทงคู่เคียงใจ จางหลิงเฮ่อ';
    const wish = data.wish || '';
    const photoMode = data.photoMode || 'มาตรฐาน';

    // บันทึกข้อมูลลงในแถวใหม่ของ Google Sheets
    sheet.appendRow([
      timestamp,
      name,
      department,
      krathongName,
      wish,
      photoMode,
      methodUsed
    ]);

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'บันทึกคำอธิษฐานลง Google Sheets เรียบร้อยแล้ว',
      timestamp: timestamp,
      method: methodUsed
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// ฟังก์ชันดึงข้อมูลคำอธิษฐานล่าสุด 50 รายการไปแสดงที่สายน้ำและตารางหน้าเว็บ
function getRecentWishes() {
  try {
    const sheet = getOrCreateSheet();
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const startRow = Math.max(2, lastRow - 49);
    const numRows = lastRow - startRow + 1;
    const values = sheet.getRange(startRow, 1, numRows, 6).getValues();

    const wishes = values.reverse().map(function(row) {
      return {
        timestamp: String(row[0]),
        name: String(row[1]),
        department: String(row[2]),
        krathongName: String(row[3]),
        wish: String(row[4]),
        photoMode: String(row[5])
      };
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      data: wishes
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

// สร้างตารางและหัวคอลัมน์อัตโนมัติหากยังไม่มี
function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    const headers = ['วัน-เวลาที่บันทึก', 'ชื่อพนักงาน', 'แผนก/ฝ่าย', 'รูปแบบกระทง', 'คำอธิษฐานจิต', 'รูปแบบภาพบนกระทง', 'ช่องทางรับข้อมูล'];
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length)
      .setBackground('#0a2540')
      .setFontColor('#ffd166')
      .setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}
📌 ๓. วิธีนำไปติดตั้งใน Google Sheets (ใช้เวลา ๑ นาที):
  1. เปิดไฟล์ Google Sheets ที่ต้องการใช้เก็บข้อมูลคำอธิษฐานของพนักงาน
  2. ไปที่เมนูด้านบน ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้วกดปุ่ม "📋 คัดลอกโค้ด Apps Script ทั้งหมด" ด้านบนไปวางแทนที่ จากนั้นกดไอคอนแผ่นดิสก์ 💾 เพื่อบันทึก
  4. กดปุ่มสีน้ำเงินมุมขวาบน การทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment)
  5. คลิกไอคอนรูปเฟือง ⚙️ เลือกประเภทเป็น เว็บแอป (Web app)
  6. ช่อง ดำเนินการในฐานะ (Execute as) เลือก ฉัน (Me) และช่อง ผู้มีสิทธิ์เข้าถึง (Who has access) เลือกเป็น ทุกคน (Anyone) เท่านั้น
  7. กดยืนยันและคัดลอกลิงก์ Web app URL (ที่ลงท้ายด้วย /exec) มาวางในช่องที่ ๑ ด้านบน แล้วกดบันทึก!
